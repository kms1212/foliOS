include(ExternalProject)

set(
    FOLIOS_SYSTEM_MANAGER_SOURCE_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/SystemManager"
    CACHE PATH
    "SystemManager user-space program source directory"
)
set(FOLIOS_SYSTEM_MANAGER_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/SystemManager")
set(FOLIOS_SYSTEM_MANAGER_APP "${FOLIOS_SYSTEM_MANAGER_BINARY_DIR}/main.app")
set(FOLIOS_SYSTEM_MANAGER_APP "${FOLIOS_SYSTEM_MANAGER_APP}" PARENT_SCOPE)

ExternalProject_Add(system_manager
    SOURCE_DIR "${FOLIOS_SYSTEM_MANAGER_SOURCE_DIR}"
    BINARY_DIR "${FOLIOS_SYSTEM_MANAGER_BINARY_DIR}"
    CMAKE_CACHE_ARGS
        -DCMAKE_MAKE_PROGRAM:FILEPATH=${CMAKE_MAKE_PROGRAM}
        -DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}
        -DCMAKE_EXPORT_COMPILE_COMMANDS:BOOL=${CMAKE_EXPORT_COMPILE_COMMANDS}
        -DCMAKE_COMPILER_LAUNCHER:FILEPATH=${CMAKE_COMPILER_LAUNCHER}
        -DCMAKE_TOOLCHAIN_FILE:FILEPATH=${CMAKE_SOURCE_DIR}/folisdk/cmake/x86_64-strata-folios.cmake
    CMAKE_ARGS
        -DBUILD_TESTING=${BUILD_TESTING}
    BUILD_ALWAYS 1
    BUILD_BYPRODUCTS "${FOLIOS_SYSTEM_MANAGER_APP}"
    INSTALL_COMMAND ""
)
