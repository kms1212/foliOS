include(ExternalProject)

ExternalProject_Add(system_manager
    SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/SystemManager"
    BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/SystemManager"
    CMAKE_CACHE_ARGS
        -DCMAKE_MAKE_PROGRAM:FILEPATH=${CMAKE_MAKE_PROGRAM}
        -DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}
        -DCMAKE_EXPORT_COMPILE_COMMANDS:BOOL=${CMAKE_EXPORT_COMPILE_COMMANDS}
        -DCMAKE_COMPILER_LAUNCHER:FILEPATH=${CMAKE_COMPILER_LAUNCHER}
        -DCMAKE_TOOLCHAIN_FILE:FILEPATH=${CMAKE_SOURCE_DIR}/folisdk/cmake/x86_64-strata-folios.cmake
    CMAKE_ARGS
        -DBUILD_TESTING=${BUILD_TESTING}
    BUILD_ALWAYS 1
    INSTALL_COMMAND ""
)
